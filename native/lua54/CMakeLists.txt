cmake_minimum_required(VERSION 3.20)
project(LumaLuaSyntax C)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS src/*.c)
list(FILTER LUA_SOURCES EXCLUDE REGEX "/(lua|luac)\\.c$")
add_library(lua54 SHARED ${LUA_SOURCES} syntax.c)
target_include_directories(lua54 PRIVATE src)
target_compile_definitions(lua54 PRIVATE LUA_BUILD_AS_DLL _CRT_SECURE_NO_WARNINGS)
set_target_properties(lua54 PROPERTIES OUTPUT_NAME "lua54-${LUMA_RUNTIME_SUFFIX}")
